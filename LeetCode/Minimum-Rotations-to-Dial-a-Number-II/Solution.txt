1class Solution {
2    public int minRotations(int n, String s) {
3        int mt=0;
4        for(int i=0;i<n;i++){
5            mt += d(i == 0 ? 0 : s.charAt(i - 1) - '0', s.charAt(i) - '0');
6        }
7        int pref=0,suf=0;
8        for(int i=n-2;i>=0;i--){
9            suf +=d(s.charAt(i+1)-'0',s.charAt(i)-'0');
10        }
11        for(int k=0;k<n;k++){
12            int prev=(k==0)?0:s.charAt(k-1)-'0';
13            mt=Math.min(mt,pref+d(prev,s.charAt(n-1)-'0')+suf);
14            pref+=d(prev,s.charAt(k)-'0');
15            if(k<n-1){
16                suf-=d(s.charAt(k+1)-'0',s.charAt(k)-'0');
17            }
18        }
19        return mt;
20        
21    }
22    private int d(int a,int b){
23        int diff=Math.abs(a-b);
24        return Math.min(diff,10-diff);
25    }
26}
